public ListNode reorder(ListNode head) {
  // 这个特判不是死代码：head==null 时下面 mid.next 会 NPE
  if (head == null || head.next == null) return head;

  // ① 找切点，切成两条独立的链
  ListNode mid    = middle(head);
  ListNode second = mid.next;
  mid.next = null;

  // ② 反转后半
  second = reverse(second);

  // ③ 交替穿插：只判短的那条
  ListNode a = head, b = second;
  while (b != null) {
    ListNode an = a.next;      // 先存，下一行就覆盖了
    ListNode bn = b.next;
    a.next = b;
    b.next = an;               // an 可能是 null，那就是链尾，正好
    a = an;
    b = bn;
  }
  return head;                 // head 没被重赋值，所以不需要 dummy
}

private ListNode middle(ListNode head) {
  ListNode slow = head, fast = head;
  // 停：fast.next==null 或 fast.next.next==null
  // 继续：把上面那句取反 —— or 变 and，== 变 !=
  while (fast.next != null && fast.next.next != null) {
    slow = slow.next;
    fast = fast.next.next;
  }
  return slow;               // 前半的最后一个
}

private ListNode reverse(ListNode head) {
  ListNode prev = null, cur = head;
  while (cur != null) {
    ListNode next = cur.next;
    cur.next = prev;
    prev = cur;
    cur = next;
  }
  return prev;
}
// TC O(n)　SC O(1)

  1. 每写一个 while，出声说完整两句：「在 ___ 继续，在 ___ 退出」。说完检查 or 有没有变 and、== 有没有变 !=。听到自己说「或者……就停」，手上就该写「并且……才继续」。
  2. 归约里哪一步只有一个动词，那一步就是会挂的那一步。这次是 "we then merge the two"——一个词，然后 bug 就在那儿。给每一步都补上「它什么时候停」。
  3. 你在 STEP 3 里自己举的那个例子，必须一路带到 STEP 5 和 STEP 7。今天你亲口说出了 1→2→3，然后在边界里没列、在 trace 里没走，判题拿它挂了你。自己举的例子是最贵的，因为它是你真正不确定的那一个。
