代码 · 这次是对的

public int closest(int[] array, int target) {
  if (array == null || array.length == 0) return -1;

  int left = 0, right = array.length - 1;

  while (left + 1 < right) {              // ← 你改对的那一处：相邻就停
    int mid = left + (right - left) / 2;  // ✓ 第三题连续说了防溢出
    if (array[mid] == target) return mid;
    else if (array[mid] < target) left = mid;   // ← 不是 mid+1：mid 可能就是答案
    else right = mid;                            // ← 同理，不是 mid-1
  }

  // 后处理：只剩两个候选，比一次距离；平局返回哪个都行
  return Math.abs(array[left] - target) <= Math.abs(array[right] - target)
       ? left : right;
}
