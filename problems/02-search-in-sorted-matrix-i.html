代码 · 一处必须改

public int[] search(int[][] matrix, int target) {
  int[] notFound = new int[]{-1, -1};

  if (matrix == null || matrix.length == 0          // ← 你漏了这个，matrix[0] 会抛异常
      || matrix[0] == null || matrix[0].length == 0) return notFound;

  int m = matrix.length, n = matrix[0].length;
  int left = 0, right = m * n - 1;

  while (left <= right) {                 // ✓ 这次说对了也写对了
    int mid = left + (right - left) / 2;  // ✓ 主动说了防溢出
    int i = mid / n, j = mid % n;         // map back: row = idx / n, col = idx mod n

    if (matrix[i][j] == target) return new int[]{i, j};
    else if (matrix[i][j] < target) left = mid + 1;   // discard the left half
    else right = mid - 1;                            // discard the right half
  }
  return notFound;
}
