收缩链
原题　数组先严格递增、再严格递减（bitonic），给一个 target。返回它出现的任意一个下标，不存在返回 −1。
收缩 ①
在无序数组里找数 → 拆成两个有序数组
整体不单调，但它是两段单调拼起来的，接缝就是峰值。知道峰在哪，就变成两次经典二分。
这一步你说得很干净，一次到位。
收缩 ②
找峰值 → 找「上升停止的第一个位置」卡在这

"And so do the right" —— 这五个字就是 Bug 1

"If a[mid] is larger or equals to left I will let left equals to mid to shrink the left half. And so do the right — if a[mid] is larger or equals to right, I will let right shrink to the mid."

"And so do the right" 描述的是「两边同样处理」，不是「否则处理右边」。你的英语和你的代码完全一致——两个平行的动作，不是二选一。所以这不是打字失误，是你当时确实这么想的。

代码 · AC 版

public int search(int[] array, int target) {
  if (array == null || array.length == 0) return -1;

  int peak = findPeak(array);                 // ① O(log n)　注意：没有 -1

  int idx = binaryAsc(array, 0, peak, target);   // ② 升序段 [0, peak]
  if (idx != -1) return idx;

  return binaryDesc(array, peak + 1, array.length - 1, target);  // ③ 降序段
}

// ① 峰 = 上升停止的第一个位置 —— First Occurrence
private int findPeak(int[] array) {
  int left = 0, right = array.length - 1;
  while (left < right) {                       // ← 收敛到一点，不是 <=
    int mid = left + (right - left) / 2;
    if (array[mid] < array[mid + 1]) left = mid + 1;  // 还在升，峰在右
    else                              right = mid;      // ← else，不是第二个 if
  }
  return left;                              // left == right == 峰值下标，无后处理
}

// ② 升序段：模版 A
private int binaryAsc(int[] a, int left, int right, int t) {
  while (left <= right) {
    int mid = left + (right - left) / 2;
    if (a[mid] == t) return mid;
    else if (a[mid] < t) left = mid + 1;
    else right = mid - 1;
  }
  return -1;
}

// ③ 降序段：一模一样，两个分支对调
private int binaryDesc(int[] a, int left, int right, int t) {
  while (left <= right) {
    int mid = left + (right - left) / 2;
    if (a[mid] == t) return mid;
    else if (a[mid] < t) right = mid - 1;  // ← 对调
    else                left = mid + 1;   // ← 对调
  }
  return -1;
}
// TC O(log n)　SC O(1)
